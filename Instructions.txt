INSTRUCCIONES DE INSTALACIÓN — RASPBERRY PI
═══════════════════════════════════════════════════════════════

# 1. Dependencias del sistema (Raspberry Pi OS / Debian)

sudo apt update
sudo apt install -y python3-venv python3-dev python3-tk \
                    wmctrl scrot \
                    libatlas-base-dev \
                    libopenblas-dev \
                    libjpeg-dev \
                    libtiff5-dev


# 2. Entorno virtual

python3 -m venv venv
source venv/bin/activate
pip install --upgrade pip

# En Windows (si pruebas allí también):
#   source venv/Scripts/activate


# 3. Requirements

pip install -r requirements.txt

# Si la línea "ncnn" falla en pip:
#   pip install ncnn-python    # nombre alternativo del paquete
# O sigue la guía oficial:
#   https://docs.ultralytics.com/guides/raspberry-pi/


═══════════════════════════════════════════════════════════════
ENTRENAMIENTO Y EXPORTACIÓN DEL MODELO
═══════════════════════════════════════════════════════════════

# 4. Entrenar el modelo (en PC potente, no en el Pi)

yolo detect train model=yolov8n.pt \
                  data=dataset/data.yaml \
                  epochs=200 \
                  imgsz=640


# 5. EXPORTAR A NCNN — CRÍTICO para velocidad en Raspberry Pi
# NCNN es 2-4x más rápido que PyTorch en ARM.
# Hacer esto en el Pi (o en el PC y copiar la carpeta resultante):

yolo export model=models/duck.pt format=ncnn

# Esto genera la carpeta: models/duck_ncnn_model/
# Asegúrate de que config.py apunte allí:
#   MODEL_PATH = "models/duck_ncnn_model"


═══════════════════════════════════════════════════════════════
AJUSTES DE RENDIMIENTO EN RASPBERRY PI
═══════════════════════════════════════════════════════════════

# En config.py, los parámetros clave para Raspberry:

#   RESIZE_WIDTH / HEIGHT = 320   -> ~50-80ms inferencia en Pi 5
#                          416   -> ~100-130ms  (mejor para targets chicos)
#                          640   -> demasiado lento (<8 FPS)

#   DEBUG = False                 -> apagar en producción, ahorra ~10% CPU

#   SHOTS_PER_TARGET = 1          -> un click por objetivo (por defecto)
#                       3         -> ráfaga de 3 clicks por objetivo
#   SHOT_BURST_DELAY = 0.04       -> 40ms entre clicks de la ráfaga

# Para verificar uso de CPU mientras corre:
#   htop
